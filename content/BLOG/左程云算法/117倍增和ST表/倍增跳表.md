```cpp
#include<bits/stdc++.h>
using namespace std;
#define LL long long 
#define vll vector<LL>


void build(vector<vll>& st,vector<int>& log2,vector<LL>& a,vector<LL>& next,int n){
    log2[0]=-1;
    for(int i=1;i<=n;i++){
        st[i][1]=next[i];
        log2[i]=log2[i>>1]+1;
    }

    for(int k=1;k<=log2[n];k++){
        for(int i=1;i<=n;i++){
            int mid=st[i][k-1];
            st[i][k]=(mid==0||mid>n)?0:st[mid][k-1];
        }
    }
}


int main(){}
```

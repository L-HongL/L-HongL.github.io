#include<bits/stdc++.h>
using namespace std;
#define LL long long

void build(vector<LL>& d,vector<LL>& a,int s,int m,int p){
    if(s==m){
        d[p]=a[s];
        return;
    }
    int u=s+((m-s)>>1);
    build(d,a,s,u,p*2);
    build(d,a,u+1,m,p*2+1);
    d[p]=d[p*2]+d[p*2+1];
}

void updata(vector<LL>& d,vector<LL>& ad,int l,int r,int s,int m,int p,LL val){
    if(l<=s&&m<=r){
        //acc(d,ad,len,s,m,p);
        d[p]+=val*(m-s+1);
        ad[p]+=val;
        return;
    }    

    int u=s+((m-s)>>1);
    if(ad[p]){
        d[p*2]+=ad[p]*(u-s+1);
        d[p*2+1]+=ad[p]*(m-u);

        ad[p*2]+=ad[p];
        ad[p*2+1]+=ad[p];
        ad[p]=0;
    }
    if(u>=l)updata(d,ad,l,r,s,u,p*2,val);
    if(u<r)updata(d,ad,l,r,u+1,m,p*2+1,val);

    d[p]=d[p*2]+d[p*2+1];
    return;
}

LL query(vector<LL>& d,vector<LL>& ad,int l,int r,int s,int m,int p){
    if(l<=s&&m<=r){
        return d[p];
    }
    int u=s+((m-s)>>1);
    if(ad[p]){
        d[p*2]+=ad[p]*(u-s+1);
        d[p*2+1]+=ad[p]*(m-u);
        ad[p*2]+=ad[p];
        ad[p*2+1]+=ad[p];
        ad[p]=0;
    }
    LL sum=0;
    if(u>=l)sum+=query(d,ad,l,r,s,u,p*2);
    if(u<r)sum+=query(d,ad,l,r,u+1,m,p*2+1);
    return sum;
}

int main(){}

#include<bits/stdc++.h>

void PrimNum(int x){
    int num=0;
    for(int i=2;i*i<=x;i++){
        if(x%i==0){
            printf("找到质因数%d",i);
            while(x%i==0)x/=i;
            num++;
        }
    }
    if(x>1){
        printf("找到质因数%d",x);
        num++;
    }
}

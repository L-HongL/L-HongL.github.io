#include<bits/stdc++.h>
using namespace std;



void SelectPrimOle(vector<int>& prim,vector<int>& isprim,int n){
    for(int i=2;i<=n;i++){//i从1开始
        if(isprim[i]==0){
            printf("find a prim %d",i);
            prim.push_back(i);
        }

        for(auto v:prim){
            if(i * v > n) break;
            isprim[v*i]=1;
            if(i%v==0)break;
        }
    }
}

int main(){}

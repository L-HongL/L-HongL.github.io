void babason(){    
    LL n,m;cin>>n>>m;
    int x,y;cin>>x>>y;
    LL ans=1e17;
    int k=lower_bound(all(a[x][y]),n)-a[x][y].begin();

        ans=min(ans,abs(n-a[x][y][k]));
    }

        ans=min(ans,abs(n-a[x][y][k-1]));
    }
    if(k-2>0){
        ans=min(ans,abs(n-a[x][y][k-2]));
    }
    cout<<ans<<"\n";
}
